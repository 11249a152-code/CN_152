EXPERIMENT 10: Local DNS Server - Name and IP Address

AIM
To write a C program to resolve a host name into an IP address and an IP address into a host name using DNS lookup functions.

PROGRAM (C - save as exp10.c)
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <netdb.h>
#include <arpa/inet.h>

int main(int argc, char *argv[]) {
    if (argc != 2) {
        printf("Usage: %s <hostname-or-ip>\n", argv[0]);
        return 1;
    }

    struct addrinfo hints, *res, *p;
    memset(&hints, 0, sizeof(hints));
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_STREAM;

    /* Try forward lookup: host name to IP address */
    int err = getaddrinfo(argv[1], NULL, &hints, &res);
    if (err == 0) {
        char ip[INET6_ADDRSTRLEN];
        puts("Host name to IP address:");
        for (p = res; p != NULL; p = p->ai_next) {
            void *addr;
            if (p->ai_family == AF_INET)
                addr = &((struct sockaddr_in *)p->ai_addr)->sin_addr;
            else if (p->ai_family == AF_INET6)
                addr = &((struct sockaddr_in6 *)p->ai_addr)->sin6_addr;
            else continue;
            inet_ntop(p->ai_family, addr, ip, sizeof(ip));
            printf("%s\n", ip);
        }
        freeaddrinfo(res);
        return 0;
    }

    /* If forward lookup fails, try reverse lookup */
    struct sockaddr_in sa;
    memset(&sa, 0, sizeof(sa));
    sa.sin_family = AF_INET;
    if (inet_pton(AF_INET, argv[1], &sa.sin_addr) == 1) {
        char host[NI_MAXHOST];
        err = getnameinfo((struct sockaddr *)&sa, sizeof(sa),
                          host, sizeof(host), NULL, 0, NI_NAMEREQD);
        if (err == 0) printf("IP address to host name: %s\n", host);
        else printf("No reverse DNS name found for %s\n", argv[1]);
    } else {
        printf("Name lookup failed: %s\n", gai_strerror(err));
    }
    return 0;
}

PROCEDURE
1. Save as exp10.c.
2. Compile: gcc exp10.c -o exp10
3. Forward lookup: ./exp10 example.com
4. Reverse lookup: ./exp10 8.8.8.8
5. The program uses the system resolver configured on the computer; it does not create a DNS server.

OUTPUT (illustrative; results depend on DNS)
$ ./exp10 example.com
Host name to IP address:
93.184.216.34
