EXPERIMENT 3: Host Name and IP Address; HTTP Client Using TCP Sockets

AIM
To write a C program to display the local host name and IP address, and to implement a simple HTTP client using TCP sockets.

PROGRAM (C - save as exp3.c)
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/socket.h>
#include <netdb.h>
#include <arpa/inet.h>

int main(int argc, char *argv[]) {
    char hostname[256];
    struct addrinfo hints, *res, *p;
    char ip[INET6_ADDRSTRLEN];

    if (gethostname(hostname, sizeof(hostname)) == 0) {
        printf("Host name: %s\n", hostname);
        memset(&hints, 0, sizeof(hints));
        hints.ai_family = AF_UNSPEC;
        hints.ai_socktype = SOCK_STREAM;
        if (getaddrinfo(hostname, NULL, &hints, &res) == 0) {
            for (p = res; p != NULL; p = p->ai_next) {
                void *addr;
                if (p->ai_family == AF_INET)
                    addr = &((struct sockaddr_in *)p->ai_addr)->sin_addr;
                else if (p->ai_family == AF_INET6)
                    addr = &((struct sockaddr_in6 *)p->ai_addr)->sin6_addr;
                else continue;
                inet_ntop(p->ai_family, addr, ip, sizeof(ip));
                printf("IP address: %s\n", ip);
            }
            freeaddrinfo(res);
        }
    }

    if (argc < 2) {
        printf("Usage: %s <hostname> [path]\n", argv[0]);
        return 0;
    }

    const char *host = argv[1];
    const char *path = (argc >= 3) ? argv[2] : "/";
    char port[] = "80";
    memset(&hints, 0, sizeof(hints));
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_STREAM;

    int err = getaddrinfo(host, port, &hints, &res);
    if (err != 0) {
        fprintf(stderr, "getaddrinfo: %s\n", gai_strerror(err));
        return 1;
    }

    int sock = -1;
    for (p = res; p != NULL; p = p->ai_next) {
        sock = socket(p->ai_family, p->ai_socktype, p->ai_protocol);
        if (sock >= 0 && connect(sock, p->ai_addr, p->ai_addrlen) == 0)
            break;
        if (sock >= 0) close(sock);
        sock = -1;
    }
    freeaddrinfo(res);
    if (sock < 0) {
        perror("connect");
        return 1;
    }

    char request[1024];
    snprintf(request, sizeof(request),
             "GET %s HTTP/1.0\r\nHost: %s\r\nConnection: close\r\n\r\n",
             path, host);
    send(sock, request, strlen(request), 0);

    char buffer[2048];
    ssize_t n;
    while ((n = recv(sock, buffer, sizeof(buffer) - 1, 0)) > 0) {
        buffer[n] = '\0';
        fputs(buffer, stdout);
    }
    close(sock);
    return 0;
}

PROCEDURE
1. Compile on Linux: gcc exp3.c -o exp3
2. Run: ./exp3 example.com /
3. The program displays the local host information and sends an HTTP GET request to the specified host.
4. This example uses plain HTTP on TCP port 80; some websites require HTTPS and will not respond correctly to this client.
